\section{Homework 1, Theory (50 min)}
\begin{task}{Task 1 --- XOR with ReLU (Problem 4)}
(a) Prove that no network without a hidden layer computes XOR: a single unit is a linear classifier
and XOR is not linearly separable (ML I, Lecture 2: the perceptron mistake bound needs a margin, and here
none exists). (b)--(d) Build the minimal ReLU network explicitly and verify all four inputs in numpy.
\end{task}
\begin{task}{Task 2 --- Xavier and He (Problem 5)}
Derive $\Var(z_k)=n_{\text{in}}\Var(w)\Var(x)$ for a linear unit with independent zero-mean weights and
inputs, hence $\Var(w)=1/n_{\text{in}}$ (Xavier). Redo it for ReLU, where half the pre-activations are
zeroed, to get $\Var(w)=2/n_{\text{in}}$ (He). Then verify in torch: initialize a 10-layer ReLU network
both ways, push a batch of $\Ncal(0,1)$ inputs through it, and plot the activation variance per layer.
\end{task}
\begin{task}{Task 3 --- Adam's bias correction (Problem 6)}
Show $\E[\vm_t]=(1-\beta_1^t)\,\vg$ under stationary gradients, hence the correction $\hat\vm_t=\vm_t/(1-\beta_1^t)$.
Compute the uncorrected bias at $t=1,10,100$ for $\beta_1=0.9$, $\beta_2=0.999$. Then run Exercise 5 of the
Homework 1 notebook (SGD, momentum, RMSprop, Adam on one problem) and relate the curves to the derivation.
\end{task}

\section{Homework 2: Convolutions (70 min)}
\begin{task}{Task 4 --- by hand, then by torch (Problems 1--6)}
\begin{enumerate}[nosep]
  \item Parameter count: a fully connected layer from $28\times28$ to $28\times28$ vs.\ one $3\times3$ convolution with 16 filters. Compute both; then read them off \texttt{sum(p.numel() for p in layer.parameters())}.
  \item $2$-D convolution of a $5\times5$ input with a $3\times3$ kernel, by hand, stride 1, no padding. Check with \texttt{torch.nn.functional.conv2d}. Note that torch computes cross-correlation; flip the kernel to see the difference.
  \item Output size $\lfloor(n+2p-k)/s\rfloor+1$: verify on four (n, k, s, p) combinations.
  \item Receptive field: two $3\times3$ convolutions give RF $5$; add a $2\times2$ max-pool (jump 2) and a third convolution; compute the RF and confirm by perturbing one input pixel and seeing which outputs change.
\end{enumerate}
\end{task}
\begin{task}{Task 5 --- pooling and architectures (Problems 7--10)}
Max vs.\ average pooling on a hand-made $4\times4$ map. LeNet-5's parameter count, layer by layer, matched
against the torch model. Then Problem 10: show that two stacked $3\times3$ convolutions have the receptive
field of one $5\times5$ with $18c^2$ instead of $25c^2$ parameters, and one more nonlinearity.
\end{task}
\begin{task}{Task 6 --- backpropagation through a convolution (Problems 11--13)}
Derive $\partial L/\partial\vK$ for a single-channel convolution: it is the cross-correlation of the input
with the upstream gradient. Implement it in numpy and compare with \texttt{autograd} on a random input.
Then the two pooling gradients: max-pool routes the gradient to the argmax (sparse); average-pool spreads
it uniformly (dense). Verify both.
\end{task}
