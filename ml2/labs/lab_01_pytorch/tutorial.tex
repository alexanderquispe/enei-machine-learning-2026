\section{Tensors (30 min)}
A tensor is a numpy array that (i) can live on a GPU and (ii) can record the operations applied to it.
\begin{task}{Task 1 --- the same computation, twice}
Redo Lab 1 of ML I in torch: the $5\times2$ design matrix, $\vX\T\vX$, and the normal-equation solution
with \texttt{torch.linalg.solve}. Confirm $(1.49,\,0.75)$. Note the three things that differ from numpy:
\texttt{dtype} defaults to \texttt{float32}, \texttt{@} works the same, and \texttt{.numpy()} converts back.
\end{task}
\begin{task}{Task 2 --- broadcasting and shapes}
Build a batch of $B=64$ inputs $\vX\in\R^{64\times p}$, a weight matrix $\vW\in\R^{p\times h}$ and a bias
$\vb\in\R^h$. Compute $\vX\vW+\vb$ and explain which axis the bias broadcasts along. Then transpose one
operand deliberately and read the error message --- you will see it again.
\end{task}

\section{Autograd (45 min)}
\begin{task}{Task 3 --- the derivative you know}
Let $f(\vw)=\tfrac12\norm{\vy-\vX\vw}^2$. Create \texttt{w = torch.zeros(p, requires\_grad=True)},
compute \texttt{f}, call \texttt{f.backward()}, and compare \texttt{w.grad} with $-\vX\T(\vy-\vX\vw)$ from
ML I, Lecture 1. They must agree to \texttt{float32} precision.
\end{task}
\begin{task}{Task 4 --- a single neuron (Exercise 2)}
Following Exercise 2 of the notebook: sigmoid neuron, BCE loss, gradient descent on
\texttt{synthetic\_2d.csv}. Before running \texttt{.backward()}, write the gradient by hand using the
identity from ML I, Lecture 2: $\partial\ell/\partial\vw=(\sigma(\vw\T\vx)-y)\,\vx$. Assert the two agree.
\end{task}
\begin{task}{Task 5 --- gradient descent, from ML I to torch (Exercise 1)}
Exercise 1: OLS vs.\ gradient descent on \texttt{diabetes.csv}. Standardize, run GD with three learning
rates, and reproduce the divergence above $2/\lambda_{\max}$ you saw in ML I --- now with
\texttt{torch.optim.SGD} doing the update. Read the source of \texttt{SGD.step} once; it is twelve lines.
\end{task}

\section{Manual Backpropagation (50 min)}
\begin{task}{Task 6 --- Exercise 3, and Problem 1 of the theory sheet}
A two-layer network with sigmoid hidden units and a sigmoid output, BCE loss. Work Problem 1(a)--(c)
on paper first (the ``beautiful cancellation'' $\partial\ell/\partial z=\hat y-y$ for sigmoid + BCE, then
the two weight gradients by the chain rule). Then implement the forward and backward passes in numpy
\emph{without} autograd, train, plot the decision boundary. Finally, rebuild the same network with
\texttt{torch.nn.Module} and check that \texttt{autograd}'s gradients match yours at the initial weights
to $10^{-6}$.
\end{task}
\begin{caution}
The most common bug: forgetting \texttt{optimizer.zero\_grad()} before \texttt{.backward()}. Gradients
\emph{accumulate} in PyTorch. Try it once deliberately and watch the loss curve.
\end{caution}
