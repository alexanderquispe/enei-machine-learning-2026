\section{Matrix Factorization by SGD (50 min)}
\begin{task}{Task 2 --- Part 1 of the notebook, Problem 17(a)--(b)}
The notebook generates a $20\times12$ panel (firms $\times$ months) of log sales with known rank-2 structure
plus noise, and masks 25\% of the entries as a staggered treatment. Implement the SGD updates from
Problem 4 with $\vU,\vV$ initialized uniformly in $[-0.5,0.5]$ and $\eta=0.03$. Sweep $K\in\{1,2,3,5,8,12\}$
and plot in-sample error $E_{\text{in}}$ (observed entries) and held-out error $E_{\text{out}}$ (masked entries).
\end{task}
\begin{task}{Task 3 --- what the curves say}
$E_{\text{in}}$ falls monotonically in $K$; $E_{\text{out}}$ turns up. Find the $K$ that minimizes $E_{\text{out}}$
and compare with the true rank. This is the bias--variance trade-off of ML I, Lecture 1, with $K$ as the
capacity knob and Eckart--Young (ML I, Lecture 6) saying what happens at $K=12$ with no missing entries.
\end{task}
\begin{task}{Task 4 --- ALS for comparison}
Implement one ALS sweep (solve for every $\vu_i$ with $\vV$ fixed, then every $\vv_j$ with $\vU$ fixed; each
is a ridge regression --- ML I, Lecture 3). Compare with SGD at the same $K$: iterations to converge,
final $E_{\text{out}}$, wall-clock time.
\end{task}

\section{Word2vec From Scratch (50 min)}
\begin{task}{Task 5 --- the gradient first (Section 2, Problem 10)}
Using $\frac{d}{dz}[-\log\sigma(z)]=\sigma(z)-1$, derive the negative-sampling gradient with respect to the
centre-word vector and to one negative sample's vector. Count the cost per update: $O((k+1)D)$ against
$O(WD)$ for the full softmax.
\end{task}
\begin{task}{Task 6 --- Part 3 of the notebook, Problem 19}
\begin{enumerate}[nosep]
  \item Tokenize \texttt{dr\_seuss.txt}; generate (centre, context) pairs with window 2.
  \item Train the skip-gram network with negative sampling in torch, $D=16$, 5 negatives, 20 epochs.
  \item Extract the embedding matrix; list the 30 most similar word pairs by cosine similarity.
  \item Reflection: which pairs make sense, which are artefacts of a 1{,}500-word corpus, and what would
        change with a corpus $10^4$ times larger.
\end{enumerate}
\end{task}
\begin{task}{Task 7 --- word2vec is matrix factorization}
Lecture 5 stated that skip-gram with negative sampling implicitly factorizes the shifted PMI matrix
(Levy \& Goldberg, 2014). Build the word--context co-occurrence matrix for the corpus, compute
$\text{PMI}-\log k$, take its rank-16 truncated SVD (ML I, Lecture 6), and compare the nearest-neighbour
lists with those of Task 6. They will not be identical; note where they agree.
\end{task}
